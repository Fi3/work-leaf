+fn osc52_copied_text(frame: &str) -> String {
+    let payload = frame
+        .strip_prefix("\u{1b}]52;c;")
+        .and_then(|rest| rest.split_once('\u{7}'))
+        .map(|(payload, _)| payload)
+        .expect("frame should start with an OSC 52 clipboard payload");
+    decode_base64(payload)
+}

+fn decode_base64(input: &str) -> String {
+    assert_eq!(input.len() % 4, 0, "base64 payload should be padded");
+    let mut bytes = Vec::new();
+    for chunk in input.as_bytes().chunks(4) {
+        let values = [
+            base64_value(chunk[0]),
+            base64_value(chunk[1]),
+            base64_value(chunk[2]),
+            base64_value(chunk[3]),
+        ];
+        let packed = (u32::from(values[0]) << 18)
+            | (u32::from(values[1]) << 12)
+            | (u32::from(values[2]) << 6)
+            | u32::from(values[3]);
+        bytes.push(((packed >> 16) & 0xff) as u8);
+        if chunk[2] != b'=' {
+            bytes.push(((packed >> 8) & 0xff) as u8);
+        }
+        if chunk[3] != b'=' {
+            bytes.push((packed & 0xff) as u8);
+        }
+    }
+    String::from_utf8(bytes).expect("OSC 52 payload should be UTF-8")
+}
+
+fn base64_value(byte: u8) -> u8 {
+    match byte {
+        b'A'..=b'Z' => byte - b'A',
+        b'a'..=b'z' => byte - b'a' + 26,
+        b'0'..=b'9' => byte - b'0' + 52,
+        b'+' => 62,
+        b'/' => 63,
+        b'=' => 0,
+        _ => panic!("invalid base64 byte {byte}"),
+    }
+}
