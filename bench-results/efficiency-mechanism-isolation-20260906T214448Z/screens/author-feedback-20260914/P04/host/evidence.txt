+fn character_selection_ranges(selection: &VisualSelection, lines: &[String]) -> Vec<(usize, usize, usize)> {
+    let (start, end) = ordered_points(selection, lines);

+fn ordered_points(selection: &VisualSelection, lines: &[String]) -> (TextPoint, TextPoint) {
+    let max_row = lines.len().saturating_sub(1);

+fn base64_encode(bytes: &[u8]) -> String {
+    const TABLE: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
+    let mut encoded = String::with_capacity(bytes.len().saturating_add(2) / 3 * 4);
+    for chunk in bytes.chunks(3) {
+        let first = chunk[0];
+        let second = *chunk.get(1).unwrap_or(&0);
+        let third = *chunk.get(2).unwrap_or(&0);
+
+        encoded.push(TABLE[usize::from(first >> 2)] as char);
+        encoded.push(TABLE[usize::from(((first & 0b0000_0011) << 4) | (second >> 4))] as char);
+        if chunk.len() > 1 {

+        } else {
+            encoded.push('=');
+        }
+        if chunk.len() > 2 {
+            encoded.push(TABLE[usize::from(third & 0b0011_1111)] as char);
+        } else {
+            encoded.push('=');
+        }
+    }

+fn character_selection_ranges(
+    selection: &VisualSelection,
+    lines: &[String],
+) -> Vec<(usize, usize, usize)> {
+    let (start, end) = ordered_points(selection, lines);
+    let mut ranges = Vec::new();
+    for row in start.row..=end.row {
+        let line_len = line_char_count(&lines[row]);
+        if line_len == 0 {
+            ranges.push((row, 0, 0));
+            continue;
+        }
+        let (range_start, range_end) = if start.row == end.row {
+            (
+                start.column.min(line_len.saturating_sub(1)),
+                end.column.min(line_len.saturating_sub(1)).saturating_add(1),

+            (start.column.min(line_len), line_len)
+        } else if row == end.row {
+            (0, end.column.min(line_len.saturating_sub(1)).saturating_add(1))
+        } else {
+            (0, line_len)
+        };
+        ranges.push((row, range_start, range_end));

+fn base64_encode(bytes: &[u8]) -> String {
+    const TABLE: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
+    let mut encoded = String::with_capacity(bytes.len().saturating_add(2) / 3 * 4);
+    for chunk in bytes.chunks(3) {
+        let first = chunk[0];
+        let second = *chunk.get(1).unwrap_or(&0);
+        let third = *chunk.get(2).unwrap_or(&0);
+
+        encoded.push(TABLE[usize::from(first >> 2)] as char);
+        encoded.push(TABLE[usize::from(((first & 0b0000_0011) << 4) | (second >> 4))] as char);
+        if chunk.len() > 1 {
+            encoded.push(TABLE[usize::from(((second & 0b0000_1111) << 2) | (third >> 6))]
+                as char);
+        } else {
+            encoded.push('=');
+        }
+        if chunk.len() > 2 {
+            encoded.push(TABLE[usize::from(third & 0b0011_1111)] as char);
+        } else {
+            encoded.push('=');
+        }
+    }
+    encoded
