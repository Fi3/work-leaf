fail_bench() { printf "%s\n" "$*" >&2; exit 86; }
setup_observer() {
  [[ -x "$observer_bin" ]] \
    || fail_bench "required benchmark observer is not executable: $observer_bin"
  local real_sh real_cargo experiment_commit real_codex
  real_sh="$(command -v sh 2>/dev/null || true)"
  [[ -n "$real_sh" ]] || fail_bench "observer could not resolve sh from PATH"
  real_cargo="$(command -v cargo 2>/dev/null || true)"
  [[ -n "$real_cargo" ]] || fail_bench "observer could not resolve cargo from PATH"
  real_codex="$bench_profiled_codex_bin"
  experiment_commit="$(git -C "$repo_root" rev-parse HEAD 2>/dev/null || printf 'unknown')"
  observer_root="$artifact_dir/observation"
  mkdir -p "$observer_root"
  local observer_identity
  observer_identity="$(stat -Lc '%d:%i' -- "$observer_root")" || fail_bench "observer directory identity unavailable"
  observer_root="$(cd -- "$observer_root" && pwd -P)" || fail_bench "observer canonical directory unavailable"
  [[ "$(stat -Lc '%d:%i' -- "$observer_root")" == "$observer_identity" ]] || fail_bench "observer canonical directory identity differs"
  "$observer_bin" init \
    --root "$observer_root" \
    --study-id "$observer_study_id" \
    --pair-id "$observer_pair_id" \
    --condition direct \
    --run-id "$run_id" \
    --real-codex "$real_codex" \
    --real-sh "$real_sh" \
    --real-cargo "$real_cargo" \
    --base-commit "$base_commit" \
    --experiment-commit "$experiment_commit" \
    --model "$bench_model" \
    --effort "$agent_reasoning_effort" \
    --require-complete-provider-usage true >/dev/null \
    || fail_bench "failed to initialize benchmark observer"
  observer_config="$observer_root/observer-config.json"
  observer_proxy_dir="$observer_root/proxy-bin"
  observer_primary_marker="$(jq -er '.primary_invocation_marker' "$observer_config")" \
    || fail_bench "observer configuration has no primary invocation marker"
  export WORK_LEAF_OBSERVER_CONFIG="$observer_config"
  export PATH="$observer_proxy_dir:$PATH"
  export WORK_LEAF_BENCH_INPUT_NEXT="$observer_proxy_dir/codex"
  agent_bin="${WORK_LEAF_BENCH_FULL_PROVIDER:?private input wrapper required}"
  observer_enabled=1
  observer_timeline condition-start "schedule=sequential"
}
observer_timeline() {
  [[ "$observer_enabled" == "1" ]] || return 0
  local event="$1"
  local detail="${2:-}"
  if ! "$observer_bin" timeline --config "$observer_config" --event "$event" --detail "$detail"; then
    log "observer timeline capture failed for $event"
  fi
  return 0
}
