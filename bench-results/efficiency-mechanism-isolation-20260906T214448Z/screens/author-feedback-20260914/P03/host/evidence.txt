+    visual_selection_ranges(selection, lines, linewise_override)

+fn visual_selection_ranges(
+    selection: &VisualSelection,
+    lines: &[String],
+    linewise_override: bool,
+) -> Vec<(usize, usize, usize)> {
+    if lines.is_empty() {
+        return Vec::new();
+    }
+
+    let anchor = clamp_selection_line(selection.anchor, lines);
+    let cursor = clamp_selection_line(selection.cursor, lines);
+    let kind = if linewise_override {
+        VisualSelectionKind::Line
+    } else {
+        selection.kind
+    };
+
+    match kind {
+        VisualSelectionKind::Line => {
+            let start = anchor.line.min(cursor.line);
+            let end = anchor.line.max(cursor.line);
+            (start..=end)
+                .map(|line| (line, 0, lines[line].chars().count()))
+                .collect()
+        }
+        VisualSelectionKind::Character => {
+            let (start, end) = ordered_positions(anchor, cursor);
+            (start.line..=end.line)
+                .map(|line| {
+                    let len = lines[line].chars().count();
+                    let start_column = if line == start.line {
+                        start.column.min(len)
+                    } else {
+                        0
+                    };
+                    let end_column = if line == end.line {
+                        end.column.saturating_add(1).min(len)
+                    } else {
+                        len
+                    };
+                    (line, start_column, end_column)
+                })
+                .collect()
+        }
+        VisualSelectionKind::Block => {
+            let start_line = anchor.line.min(cursor.line);
+            let end_line = anchor.line.max(cursor.line);
+            let start_column = anchor.column.min(cursor.column);
+            let end_column = anchor.column.max(cursor.column).saturating_add(1);
+            (start_line..=end_line)
+                .map(|line| {
+                    let len = lines[line].chars().count();
+                    (line, start_column.min(len), end_column.min(len))
+                })
+                .collect()
+        }
+    }
+}

+fn base64_encode(input: &[u8]) -> String {

+    let mut output = String::with_capacity(((input.len() + 2) / 3) * 4);

+        let first = chunk[0];
+        let second = *chunk.get(1).unwrap_or(&0);
+        let third = *chunk.get(2).unwrap_or(&0);
+        output.push(TABLE[(first >> 2) as usize] as char);
+        output.push(TABLE[(((first & 0b0000_0011) << 4) | (second >> 4)) as usize] as char);
+        if chunk.len() > 1 {

+        } else {

+fn visual_selection_ranges(
+    selection: &VisualSelection,
+    lines: &[String],
+    linewise_override: bool,
+) -> Vec<(usize, usize, usize)> {
+    if lines.is_empty() {
+        return Vec::new();
+    }
+
+    let anchor = clamp_selection_line(selection.anchor, lines);
+    let cursor = clamp_selection_line(selection.cursor, lines);
+    let kind = if linewise_override {
+        VisualSelectionKind::Line
+    } else {
+        selection.kind
+    };
+
+    match kind {
+        VisualSelectionKind::Line => {
+            let start = anchor.line.min(cursor.line);
+            let end = anchor.line.max(cursor.line);
+            (start..=end)
+                .map(|line| (line, 0, lines[line].chars().count()))
+                .collect()
+        }
+        VisualSelectionKind::Character => {
+            let (start, end) = ordered_positions(anchor, cursor);

+                        start.column.min(len)
+                    } else {
+                        0
+                    };
+                    let end_column = if line == end.line {
+                        end.column.saturating_add(1).min(len)
+                    } else {
+                        len
+                    };
+                    (line, start_column, end_column)
+                })
+                .collect()
+        }
+        VisualSelectionKind::Block => {
+            let start_line = anchor.line.min(cursor.line);
+            let end_line = anchor.line.max(cursor.line);
+            let start_column = anchor.column.min(cursor.column);
+            let end_column = anchor.column.max(cursor.column).saturating_add(1);

+fn base64_encode(input: &[u8]) -> String {
+    const TABLE: &[u8; 64] =
+        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
+    let mut output = String::with_capacity(((input.len() + 2) / 3) * 4);
+    for chunk in input.chunks(3) {
+        let first = chunk[0];
+        let second = *chunk.get(1).unwrap_or(&0);
+        let third = *chunk.get(2).unwrap_or(&0);
+        output.push(TABLE[(first >> 2) as usize] as char);
+        output.push(TABLE[(((first & 0b0000_0011) << 4) | (second >> 4)) as usize] as char);
+        if chunk.len() > 1 {
+            output.push(TABLE[(((second & 0b0000_1111) << 2) | (third >> 6)) as usize] as char);
+        } else {
+            output.push('=');
+        }
+        if chunk.len() > 2 {
+            output.push(TABLE[(third & 0b0011_1111) as usize] as char);
+        } else {
+            output.push('=');
+        }
+    }
+    output
