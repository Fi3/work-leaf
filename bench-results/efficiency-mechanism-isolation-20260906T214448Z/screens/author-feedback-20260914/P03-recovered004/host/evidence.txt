+fn selected_range_for_line(
+    selection: VisualSelection,
+    line_index: usize,
+    line_len: usize,
+) -> Option<(usize, usize)> {
+    let (start_line, end_line) = selected_line_bounds(selection);
+    if line_index < start_line || line_index > end_line {
+        return None;
+    }

+        UiVisualMode::Line => Some((0, line_len)),

+            let start = selection.anchor.column.min(selection.cursor.column);
+            let end = selection.anchor.column.max(selection.cursor.column);
+            exclusive_column_range(start, end, line_len)

+            } else if line_index == end.line {
+                exclusive_column_range(0, end.column, line_len)
+            } else {
+                Some((0, line_len))
+            }
+        }
+    }

+fn exclusive_column_range(start: usize, end: usize, line_len: usize) -> Option<(usize, usize)> {
+    if line_len == 0 {
+        return Some((0, 0));
+    }
+    if start >= line_len {
+        return Some((line_len, line_len));
+    }
+    let end = end.min(line_len - 1);

+        return None;
+    }

+    } else {

+    }

+fn osc52_clipboard_sequence(text: &str) -> String {
+    format!("\u{1b}]52;c;{}\u{7}", base64_encode(text.as_bytes()))

+fn base64_encode(bytes: &[u8]) -> String {
+    const TABLE: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
+    let mut encoded = String::with_capacity(bytes.len().div_ceil(3) * 4);
+    for chunk in bytes.chunks(3) {
+        let first = chunk[0];
+        let second = chunk.get(1).copied().unwrap_or(0);
+        let third = chunk.get(2).copied().unwrap_or(0);
+        encoded.push(TABLE[(first >> 2) as usize] as char);
+        encoded.push(TABLE[(((first & 0b0000_0011) << 4) | (second >> 4)) as usize] as char);
+        if chunk.len() > 1 {
+            encoded.push(TABLE[(((second & 0b0000_1111) << 2) | (third >> 6)) as usize] as char);
+        } else {
+            encoded.push('=');
+        }
+        if chunk.len() > 2 {
+            encoded.push(TABLE[(third & 0b0011_1111) as usize] as char);
+        } else {
+            encoded.push('=');
+        }
+    }
+    encoded

+fn clamp_text_position(
+    mut position: TextPosition,
+    lines: &[String],
+    mode: UiVisualMode,
+) -> TextPosition {
+    let max_line = lines.len().saturating_sub(1);

+        position.column = 0;
+        return position;

+fn selected_text(
+    selection: VisualSelection,
+    lines: &[String],
+    copy_kind: VisualCopyKind,
+) -> String {
+    if lines.is_empty() {
+        return String::new();
+    }
+    let (start_line, end_line) = selected_line_bounds(selection);
+    let start_line = start_line.min(lines.len() - 1);
+    let end_line = end_line.min(lines.len() - 1);
+    let mut selected = Vec::new();
+    for line_index in start_line..=end_line {
+        let line = &lines[line_index];
+        if copy_kind == VisualCopyKind::Lines {
+            selected.push(line.clone());
+            continue;
+        }
+        if let Some((start, end)) =
+            selected_range_for_line(selection, line_index, line_char_count(line))
+        {
+            selected.push(slice_chars(line, start, end));
+        }
+    }
+    selected.join("\n")
+}

+fn selected_line_bounds(selection: VisualSelection) -> (usize, usize) {
+    if selection.anchor.line <= selection.cursor.line {
+        (selection.anchor.line, selection.cursor.line)
+    } else {
+        (selection.cursor.line, selection.anchor.line)
+    }
+}

+fn selected_range_for_line(
+    selection: VisualSelection,
+    line_index: usize,
+    line_len: usize,
+) -> Option<(usize, usize)> {
+    let (start_line, end_line) = selected_line_bounds(selection);
+    if line_index < start_line || line_index > end_line {
+        return None;
+    }
+
+    match selection.mode {
+        UiVisualMode::Line => Some((0, line_len)),
+        UiVisualMode::Block => {
+            let start = selection.anchor.column.min(selection.cursor.column);
+            let end = selection.anchor.column.max(selection.cursor.column);
+            exclusive_column_range(start, end, line_len)
+        }
+        UiVisualMode::Character => {
+            let (start, end) = ordered_positions(selection.anchor, selection.cursor);
+            if start.line == end.line {
+                return exclusive_column_range(start.column, end.column, line_len);
+            }
+            if line_index == start.line {
+                exclusive_column_range(start.column, line_len.saturating_sub(1), line_len)
+            } else if line_index == end.line {
+                exclusive_column_range(0, end.column, line_len)
+            } else {
+                Some((0, line_len))
+            }
+        }
+    }
+}
+
+fn exclusive_column_range(start: usize, end: usize, line_len: usize) -> Option<(usize, usize)> {
+    if line_len == 0 {
+        return Some((0, 0));
+    }
+    if start >= line_len {
+        return Some((line_len, line_len));
+    }
+    let end = end.min(line_len - 1);
+    if start > end {
+        return None;
+    }
+    Some((start, end + 1))
+}
+
+fn ordered_positions(left: TextPosition, right: TextPosition) -> (TextPosition, TextPosition) {
+    if left.line < right.line || (left.line == right.line && left.column <= right.column) {
+        (left, right)
+    } else {
+        (right, left)
+    }
+}

+    text.chars()
+        .skip(start)
+        .take(end.saturating_sub(start))
+        .collect()
+}

+fn osc52_clipboard_sequence(text: &str) -> String {
+    format!("\u{1b}]52;c;{}\u{7}", base64_encode(text.as_bytes()))
+}

+fn base64_encode(bytes: &[u8]) -> String {
+    const TABLE: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
+    let mut encoded = String::with_capacity(bytes.len().div_ceil(3) * 4);
+    for chunk in bytes.chunks(3) {
+        let first = chunk[0];
+        let second = chunk.get(1).copied().unwrap_or(0);
+        let third = chunk.get(2).copied().unwrap_or(0);
+        encoded.push(TABLE[(first >> 2) as usize] as char);
+        encoded.push(TABLE[(((first & 0b0000_0011) << 4) | (second >> 4)) as usize] as char);
+        if chunk.len() > 1 {
+            encoded.push(TABLE[(((second & 0b0000_1111) << 2) | (third >> 6)) as usize] as char);
+        } else {
+            encoded.push('=');
+        }
+        if chunk.len() > 2 {
+            encoded.push(TABLE[(third & 0b0011_1111) as usize] as char);
+        } else {
+            encoded.push('=');
+        }
+    }
+    encoded
+}

+fn clamp_text_position(
+    mut position: TextPosition,
+    lines: &[String],
+    mode: UiVisualMode,
+) -> TextPosition {
+    let max_line = lines.len().saturating_sub(1);
+    position.line = position.line.min(max_line);
+    if mode == UiVisualMode::Line {
+        position.column = 0;
+        return position;
+    }
+    let line = lines.get(position.line).map(String::as_str).unwrap_or("");
+    position.column = position.column.min(max_selectable_column(line));
+    position

+fn selected_text(
+    selection: VisualSelection,
+    lines: &[String],
+    copy_kind: VisualCopyKind,
+) -> String {
+    if lines.is_empty() {
+        return String::new();
+    }
+    let (start_line, end_line) = selected_line_bounds(selection);
+    let start_line = start_line.min(lines.len() - 1);
+    let end_line = end_line.min(lines.len() - 1);
+    let mut selected = Vec::new();
+    for line_index in start_line..=end_line {
+        let line = &lines[line_index];
+        if copy_kind == VisualCopyKind::Lines {
+            selected.push(line.clone());
+            continue;
+        }
+        if let Some((start, end)) =
+            selected_range_for_line(selection, line_index, line_char_count(line))
+        {
+            selected.push(slice_chars(line, start, end));
+        }
+    }
+    selected.join("\n")
+}

+fn selected_line_bounds(selection: VisualSelection) -> (usize, usize) {
+    if selection.anchor.line <= selection.cursor.line {
+        (selection.anchor.line, selection.cursor.line)
+    } else {
+        (selection.cursor.line, selection.anchor.line)
+    }
+}
+
+fn selected_range_for_line(
+    selection: VisualSelection,
+    line_index: usize,
+    line_len: usize,
+) -> Option<(usize, usize)> {
+    let (start_line, end_line) = selected_line_bounds(selection);
+    if line_index < start_line || line_index > end_line {
+        return None;
+    }
+
+    match selection.mode {
+        UiVisualMode::Line => Some((0, line_len)),
+        UiVisualMode::Block => {
+            let start = selection.anchor.column.min(selection.cursor.column);
+            let end = selection.anchor.column.max(selection.cursor.column);
+            exclusive_column_range(start, end, line_len)
+        }
+        UiVisualMode::Character => {
+            let (start, end) = ordered_positions(selection.anchor, selection.cursor);
+            if start.line == end.line {
+                return exclusive_column_range(start.column, end.column, line_len);
+            }
+            if line_index == start.line {
+                exclusive_column_range(start.column, line_len.saturating_sub(1), line_len)
+            } else if line_index == end.line {
+                exclusive_column_range(0, end.column, line_len)
+            } else {
+                Some((0, line_len))
+            }
+        }
+    }
+}
+
+fn exclusive_column_range(start: usize, end: usize, line_len: usize) -> Option<(usize, usize)> {
+    if line_len == 0 {
+        return Some((0, 0));
+    }
+    if start >= line_len {
+        return Some((line_len, line_len));
+    }
+    let end = end.min(line_len - 1);
+    if start > end {
+        return None;
+    }
+    Some((start, end + 1))
+}
+
+fn ordered_positions(left: TextPosition, right: TextPosition) -> (TextPosition, TextPosition) {
+    if left.line < right.line || (left.line == right.line && left.column <= right.column) {
+        (left, right)
+    } else {
+        (right, left)
+    }
+}

+fn slice_chars(text: &str, start: usize, end: usize) -> String {
+    text.chars()
+        .skip(start)
+        .take(end.saturating_sub(start))
+        .collect()
+}

+fn osc52_clipboard_sequence(text: &str) -> String {
+    format!("\u{1b}]52;c;{}\u{7}", base64_encode(text.as_bytes()))

+fn base64_encode(bytes: &[u8]) -> String {
+    const TABLE: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
+    let mut encoded = String::with_capacity(bytes.len().div_ceil(3) * 4);
+    for chunk in bytes.chunks(3) {
+        let first = chunk[0];
+        let second = chunk.get(1).copied().unwrap_or(0);
+        let third = chunk.get(2).copied().unwrap_or(0);
+        encoded.push(TABLE[(first >> 2) as usize] as char);
+        encoded.push(TABLE[(((first & 0b0000_0011) << 4) | (second >> 4)) as usize] as char);
+        if chunk.len() > 1 {
+            encoded.push(TABLE[(((second & 0b0000_1111) << 2) | (third >> 6)) as usize] as char);
+        } else {
+            encoded.push('=');
+        }
+        if chunk.len() > 2 {
+            encoded.push(TABLE[(third & 0b0011_1111) as usize] as char);
+        } else {
+            encoded.push('=');
+        }
+    }
+    encoded
