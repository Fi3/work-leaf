+fn character_selection_ranges(
+    anchor: SelectionPoint,
+    cursor: SelectionPoint,
+    lines: &[String],
+) -> Vec<(usize, usize, usize)> {
+    let (start, end) = if point_le(anchor, cursor) {
+        (anchor, cursor)
+    } else {
+        (cursor, anchor)
+    };
+    (start.row..=end.row)
+        .map(|row| {
+            let len = char_len(&lines[row]);
+            if len == 0 {
+                return (row, 0, 0);
+            }
+            if start.row == end.row {
+                let start_column = start.column.min(len - 1);
+                let end_column = end.column.min(len - 1).saturating_add(1);
+                return (row, start_column, end_column.max(start_column));
+            }
+            if row == start.row {
+                let start_column = start.column.min(len - 1);
+                return (row, start_column, len);
+            }
+            if row == end.row {
+                let end_column = end.column.min(len - 1).saturating_add(1);
+                return (row, 0, end_column);
